    <script>
        // 1. Calculateur automatique de tranche 50%
        function calculerMontant() {
            let total = parseFloat(document.getElementById('totalAmount').value);
            let tranche = document.getElementById('tranche').value;
            let resultInput = document.getElementById('payAmount');

            if (!isNaN(total) && total > 0) {
                if (tranche === 'acompte' || tranche === 'solde') {
                    let verse = total * 0.50;
                    resultInput.value = verse.toLocaleString('fr-FR') + ' FCFA (Tranche de 50%)';
                } else if (tranche === 'autre') {
                    resultInput.value = total.toLocaleString('fr-FR') + ' FCFA (100%)';
                } else {
                    resultInput.value = '';
                }
            } else {
                resultInput.value = '';
            }
        }

        // 2. Soumission du devis
        function soumettreDevis(event) {
            event.preventDefault();
            document.getElementById('devisSuccess').style.display = 'block';
        }

        // 3. Validation du règlement
        function validerPaiement(event) {
            event.preventDefault();
            let payer = document.getElementById('payerName').value;
            let ref = document.getElementById('contractRef').value;
            let montant = document.getElementById('payAmount').value;

            let successBox = document.getElementById('paiementSuccess');
            successBox.innerHTML = '✅ Reçu enregistré pour <strong>' + payer + '</strong> (Réf Contrat : ' + ref + '). Montant validé : <strong>' + montant + '</strong>. L\'information a été transmise à l\'administration SERVICE JRPMT.';
            successBox.style.display = 'block';
        }

        // 4. Génération dynamique de l'email pro
        document.getElementById('empName').addEventListener('input', function() {
            let input = this.value.trim().toLowerCase();
            let parts = input.split(' ');
            if (parts.length >= 2) {
                let initial = parts[0].charAt(0);
                let nom = parts.slice(1).join('');
                document.getElementById('emailResult').value = initial + '.' + nom + '@servicejrpmt.ml';
            } else if (parts.length === 1 && parts[0] !== '') {
                document.getElementById('emailResult').value = parts[0] + '@servicejrpmt.ml';
            } else {
                document.getElementById('emailResult').value = '';
            }
        });

        function validerCompte() {
            let mail = document.getElementById('emailResult').value;
            if (mail !== '') {
                document.getElementById('statusMessage').innerHTML = '✅ L\'adresse officielle <strong>' + mail + '</strong> est validée et réservée sur le serveur d\'entreprise.';
            }
        }
    </script>
